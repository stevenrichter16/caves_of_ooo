        static object PhysicalLightingSnapshot(Zone zone)
        {
            // Unity Color exposes recursive computed properties to Json.NET;
            // report only detached observed channels, without altering lighting.
            var tint=zone.AmbientTint;
            return new{ambientLevel=zone.AmbientLevel,ambientTint=new{tint.r,tint.g,tint.b,tint.a}};
        }
        Entity PhysicalPlace(Zone zone,string blueprint,int x,int y)
        {var e=_context.Factory.CreateEntity(blueprint);Require(e!=null&&zone.AddEntity(e,x,y),"factory controlled placement "+blueprint);return e;}
        void PhysicalPursuerMoved(Entity entity,Zone zone,int ox,int oy,int nx,int ny,bool forced)
        {
            if(entity!=_physicalPursuer||zone!=_physicalArena)return;
            _physicalMoves++;bool legal=!forced&&Math.Max(Math.Abs(nx-ox),Math.Abs(ny-oy))==1&&zone.GetEntityPosition(entity)==(nx,ny)
                &&zone.GetCell(nx,ny)!=null&&!zone.GetCell(nx,ny).BlocksMovement(entity);
            if(!legal)_physicalMoveError="Pursuer movement was forced, nonadjacent or entered a physical obstacle.";
            if(nx==37&&(ny<8||ny>16))_physicalCrossedEnd=true;
            _observations.Add(new{phase="actual-controlled-pursuer-move",owner=entity.ID,oldCell=new[]{ox,oy},newCell=new[]{nx,ny},forced,legal,tick=Tick});
        }
        void RecordPhysicalPursuit(string label,int waits)
        {
            var brain=_physicalPursuer.GetPart<BrainPart>();var goals=brain.GetGoalsSnapshot().ToArray();
            _physicalCurrentPursuit=ReferenceEquals(brain.Target,Player)&&goals.OfType<KillGoal>().Any(g=>ReferenceEquals(g.Target,Player));
            _physicalAcquired|=_physicalCurrentPursuit;
            _observations.Add(new{phase=label,waits,tick=Tick,player=new{Player.ID,position=Zone.GetEntityPosition(Player),energy=Energy,hp=Player.GetStatValue("Hitpoints"),speed=Player.GetStatValue("Speed")},
                pursuer=new{_physicalPursuer.ID,position=Zone.GetEntityPosition(_physicalPursuer),energy=_input.TurnManager.GetEnergy(_physicalPursuer),hp=_physicalPursuer.GetStatValue("Hitpoints"),target=brain.Target?.ID,
                    goals=goals.Select(g=>new{type=g.GetType().Name,g.Age}).ToArray()},acquired=_physicalAcquired,currentPursuit=_physicalCurrentPursuit,usedOpenEnd=_physicalCrossedEnd});WriteReport();
        }
        static int PhysicalBypassLength(Zone zone,Entity actor,Entity target)
        {
            var start=zone.GetEntityCell(actor);var finish=zone.GetEntityCell(target);if(start==null||finish==null)return -1;
            var pending=new Queue<(Cell cell,int steps)>();var seen=new HashSet<Cell>{start};pending.Enqueue((start,0));
            while(pending.Count>0){var row=pending.Dequeue();if(row.cell==finish)return row.steps;
                foreach(var d in HaulDirections){var c=zone.GetCell(row.cell.X+d.x,row.cell.Y+d.y);if(c==null||!seen.Add(c))continue;
                    if(c.Occupants.Any(e=>e!=actor&&e!=target&&(e.HasTag("Solid")||e.GetPart<PhysicsPart>()?.Solid==true||e.GetPart<DoorPart>()?.IsClosed==true)))continue;
                    pending.Enqueue((c,row.steps+1));}}
            return -1;
        }
        void StopPhysicalPursuitObservation()
        {
            if(!_physicalObserving)return;_physicalObserving=false;EntityVisualHooks.MovedCallback-=PhysicalPursuerMoved;
            if(_input!=null&&_physicalLoad!=null&&ReferenceEquals(DragSystem.GetDragged(Player),_physicalLoad))DragSystem.Release(Player);
            // The arena belongs to this disposable isolated Play world. Teardown
            // discards it; do not rewind live tick/energy or rebuild the old world.
        }
