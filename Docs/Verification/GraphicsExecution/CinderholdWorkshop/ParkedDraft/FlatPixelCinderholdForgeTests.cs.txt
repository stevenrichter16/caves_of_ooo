    public sealed class FlatPixelCinderholdForgeTests
    {
        const int X = 55, Y = 5;
        const BindingFlags I = BindingFlags.Instance | BindingFlags.Public | BindingFlags.NonPublic;
        DensityLootTestScope content;
        OverworldZoneManager manager;
        Zone zone;
        Entity forge, floor;
        FlatPixelGreyboxCatalog world;
        Texture2D Atlas => Resources.Load<Texture2D>(FlatPixelEastKitchenCatalog.ResourcePath);
        [SetUp] public void Setup()
        {
            content = new DensityLootTestScope();
            manager = OverworldZoneManager.CreateDetached(content.Factory, 704860946, true);
            zone = Attach("Overworld.6.6.0"); floor = Add("StoneFloor"); forge = Add("TinkersForge");
            world = new FlatPixelGreyboxCatalog();
        }
        Zone Attach(string id)
        { var z = new Zone(id); manager.SetActiveZone(z); foreach (var c in z.Cells) c.IsVisible = c.Explored = true; return z; }
        Entity Add(string name)
        { var e = content.Factory.CreateEntity(name); Assert.NotNull(e); Assert.True(zone.AddEntity(e,X,Y)); return e; }
        [TearDown] public void Cleanup() { world?.Dispose(); content?.Dispose(); }
        FlatPixelGreyboxCell View()
        { Assert.True(world.TryResolvePresentation(zone,X,Y,out var v)); return v; }
        bool ReaderNative(int x = X, int y = Y)
            => world.TryGetPresentationSubjectSprite(zone,forge,x,y,out var s,out _) && s.texture == Atlas;
        void Native()
        {
            var v = View(); Assert.AreSame(forge,v.Owner); Assert.True(v.NativeForeground,"Cinderhold forge body is still a glyph");
            Assert.False(v.BlocksMovement); Assert.True(v.LocalGround); Assert.True(v.PreserveRenderColorCue); Assert.True(ReaderNative());
            Assert.AreSame(Atlas,v.ForegroundTile.sprite.texture);
            Assert.AreEqual(new Rect(0,0,16,24),v.ForegroundTile.sprite.rect);
            Assert.AreEqual(new Vector2(8,12),v.ForegroundTile.sprite.pivot);
            Assert.AreEqual(16,v.ForegroundTile.sprite.pixelsPerUnit);
            Assert.AreEqual(Tile.ColliderType.None,v.ForegroundTile.colliderType);
        }
        void NoForge()
        {
            if(world.TryResolvePresentation(zone,X,Y,out var v))
                Assert.False(v.NativeForeground && v.ForegroundTile?.sprite?.texture == Atlas && v.ForegroundTile.sprite.rect.x == 0);
            Assert.False(ReaderNative());
        }
        [Test] public void PublicWorldAndReaderUseCachedUnlitBodyWithoutChangingServiceOrGround()
        {
            string parts = string.Join("|",forge.Parts.Select(p=>JsonUtility.ToJson(p)));
            string state = zone.TileState.ToSaveString(); int version = zone.EntityVersion;
            Native(); var first=View().ForegroundTile; Native(); Assert.AreSame(first,View().ForegroundTile);
            Assert.AreSame(floor,zone.GetCell(X,Y).Objects[0]); Assert.AreEqual(2,zone.GetCell(X,Y).Objects.Count);
            Assert.AreEqual(parts,string.Join("|",forge.Parts.Select(p=>JsonUtility.ToJson(p))));
            Assert.AreEqual(version,zone.EntityVersion); Assert.AreEqual(state,zone.TileState.ToSaveString());
            Assert.IsNull(forge.GetPart<ThermalPart>()); Assert.IsNull(forge.GetPart<LightSourcePart>());
        }
        [Test] public void GeneratedRealWorkshopForgeUsesApprovedBody()
        {
            manager.CachedZones.Remove(zone.ZoneID); zone=manager.GetZone(zone.ZoneID); manager.SetActiveZone(zone);
            foreach(var c in zone.Cells) c.IsVisible=c.Explored=true;
            forge=zone.GetAllEntities().Single(e=>e.BlueprintName=="TinkersForge");
            Assert.AreEqual(X,zone.GetEntityCell(forge).X); Assert.AreEqual(Y,zone.GetEntityCell(forge).Y); Native();
        }
        [TestCase("glyph")][TestCase("color")][TestCase("background")][TestCase("detail")][TestCase("tilecolor")]
        [TestCase("tile")][TestCase("visual")][TestCase("variant")][TestCase("glyphVariants")][TestCase("layer")]
        [TestCase("hidden")][TestCase("solid")][TestCase("takeable")][TestCase("inventory")][TestCase("equipped")]
        [TestCase("name")][TestCase("tag")][TestCase("extraTag")][TestCase("nullTags")]
        [TestCase("thermal")][TestCase("light")][TestCase("repair")][TestCase("destructible")][TestCase("service")]
        [TestCase("zeroStack")][TestCase("multiCell")]
        public void ChangedMeaningKeepsFallback(string fault)
        {
            Native(); var r=forge.GetPart<RenderPart>(); var p=forge.GetPart<PhysicsPart>();
            switch(fault)
            {
                case "glyph": r.RenderString="!"; break; case "color": r.ColorString="&G"; break;
                case "background": r.BackgroundColor="^r"; break; case "detail": r.DetailColor="&G"; break;
                case "tilecolor": r.TileColor="&G"; break; case "tile": r.Tile="custom"; break;
                case "visual": r.VisualID="custom"; break; case "variant": r.VisualVariant="custom"; break;
                case "glyphVariants": r.GlyphVariants="nN"; break; case "layer": r.RenderLayer=2; break;
                case "hidden": r.Visible=false; break; case "solid": p.Solid=true; break;
                case "takeable": p.Takeable=true; break; case "inventory": p.InInventory=new Entity(); break;
                case "equipped": p.Equipped=new Entity(); break; case "name": forge.BlueprintName="other"; break;
                case "tag": forge.Tags.Remove("Weaponcraft"); break; case "extraTag": forge.Tags["Restable"]=""; break;
                case "nullTags":
                    // Public fallback assumes a tag dictionary before reaching this adapter.
                    // Exercise the adapter's corruption guard without claiming to repair that older contract.
                    forge.Tags=null;
                    using(var art=new FlatPixelEastKitchenCatalog(Atlas,null))
                        Assert.False((bool)typeof(FlatPixelEastKitchenCatalog).GetMethod("TryResolveOrdinaryForge",I)
                            .Invoke(art,new object[]{forge,null}));
                    return;
                case "thermal": forge.AddPart(new ThermalPart()); break;
                case "light": forge.AddPart(new LightSourcePart()); break; case "repair": forge.AddPart(new RepairablePart()); break;
                case "destructible": forge.AddPart(new DestructiblePart()); break; case "service": forge.AddPart(new ContainerPart()); break;
                case "zeroStack": forge.AddPart(new StackerPart{StackCount=0}); break;
                case "multiCell": forge.AddPart(new SpatialFootprintPart{CellsRaw="0,0"}); Assert.True(zone.TryChangeFootprint(forge,"0,0;1,0")); break;
            }
            NoForge();
        }
        [TestCase("Render")][TestCase("Physics")][TestCase("Forge")][TestCase("Examinable")]
        public void RequiredPartsMustBeUniqueAndOwned(string kind)
        {
            Native(); var part=forge.GetPart(kind); part.ParentEntity=new Entity(); NoForge(); part.ParentEntity=forge; Native();
            var duplicate=(Part)Activator.CreateInstance(part.GetType()); forge.AddPart(duplicate); NoForge();
            forge.RemovePart(duplicate); Native(); forge.RemovePart(part); NoForge();
        }
        [TestCase("Stacker")][TestCase("SpatialFootprint")][TestCase("StatusEffects")]
        public void OptionalPartsMustBeUniqueAndOwned(string kind)
        {
            Native(); Part p=kind=="Stacker"?(Part)new StackerPart{StackCount=1}:kind=="SpatialFootprint"?
                new SpatialFootprintPart{CellsRaw="0,0"}:(Part)new StatusEffectsPart();
            forge.AddPart(p); Native(); p.ParentEntity=new Entity(); NoForge(); p.ParentEntity=forge; Native();
            // AddPart correctly forbids a second physical footprint. Inject the
            // malformed list directly to exercise this read-only corruption guard.
            var duplicate=(Part)Activator.CreateInstance(p.GetType()); duplicate.ParentEntity=forge;
            forge.Parts.Add(duplicate); NoForge();
        }

        [Test] public void CoverAndCurrentReaderAgreeWithoutShowingHiddenOrMovedOwner()
        {
            Native(); var doc=ExamineDocumentBuilder.BuildWorld(forge,zone,zone.GetCell(X,Y),null);
            using(var paint=new FlatPixelSharedFurnitureTests.Paint(world,zone))
            using(var portraits=(ExaminePortraitRenderer)Activator.CreateInstance(typeof(ExaminePortraitRenderer),I,null,
                new object[]{(Action<Camera,RenderTexture>)((c,t)=>{})},null))
            {
                paint.Step(); Assert.AreSame(Atlas,paint.At(X,Y).sprite.texture);
                var cover=Add("Player"); paint.Dirty(X,Y); paint.Step(); Assert.AreSame(cover,View().Owner);
                Assert.AreSame(Atlas,portraits.GetPortrait(doc,paint.Renderer).Texture);
                zone.RemoveEntity(cover); zone.GetCell(X,Y).IsVisible=false; paint.Dirty(X,Y); paint.Step(); NoForge();
                Assert.IsNull(portraits.GetPortrait(doc,paint.Renderer).Texture);
                zone.GetCell(X,Y).Explored=false; Assert.False(world.TryResolvePresentation(zone,X,Y,out _));
                zone.GetCell(X,Y).IsVisible=zone.GetCell(X,Y).Explored=true; Native();
                Assert.True(zone.MoveEntity(forge,X+1,Y)); NoForge(); Assert.False(ExamineDocumentBuilder.IsCurrentSubject(doc));
                zone.GetCell(X,Y).Objects.Add(forge); NoForge(); zone.GetCell(X,Y).Objects.Remove(forge);
                zone.RemoveEntity(forge); Assert.IsNull(portraits.GetPortrait(doc,paint.Renderer).Texture);
            }
        }
        [TestCase(false)][TestCase(true)] public void ExplicitMissingAtlasDoesNotReloadResources(bool malformed)
        {
            world.Dispose(); world=new FlatPixelGreyboxCatalog((Texture2D)null);
            var texture=malformed?new Texture2D(1,1):null;
            try
            {
                var east=new FlatPixelEastKitchenCatalog(texture,null);
                typeof(FlatPixelGreyboxCatalog).GetField("_eastKitchenArt",I).SetValue(world,east); NoForge();
            }
            finally { world.Dispose(); if(texture!=null) Object.DestroyImmediate(texture); }
        }
        [Test] public void BorrowedInjectionIsCachedDisposedAndNeverModifiesTexture()
        {
            world.Dispose(); world=new FlatPixelGreyboxCatalog((Texture2D)null); var texture=new Texture2D(288,24);
            try
            {
                var east=new FlatPixelEastKitchenCatalog(texture,null);
                typeof(FlatPixelGreyboxCatalog).GetField("_eastKitchenArt",I).SetValue(world,east);
                var first=View(); Assert.True(first.NativeForeground); Assert.AreSame(texture,first.ForegroundTile.sprite.texture);
                Assert.AreSame(first.ForegroundTile,View().ForegroundTile); world.Dispose(); world.Dispose();
                Assert.True(first.ForegroundTile==null); Assert.True(texture!=null);
                Assert.AreEqual(FilterMode.Bilinear,texture.filterMode); Assert.AreEqual(TextureWrapMode.Repeat,texture.wrapMode);
            }
            finally { Object.DestroyImmediate(texture); }
        }
