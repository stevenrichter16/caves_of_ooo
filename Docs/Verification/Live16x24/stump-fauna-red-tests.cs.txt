                zone.GetCell(X, Y).IsVisible = false; paint.Dirty(X, Y); paint.Step();

        [TestCase("CascadeFather")] [TestCase("GlasspaneFrog")] [TestCase("YellowfootWayfarer")] [TestCase("Wardline")] [TestCase("SariSnake")]
        public void AppliedDamageKeepsOneRenderEventAndThreeNativeHurtPaintsWithoutReaderOrNeighborConsumption(string species)
        {
            var owner = Add(species); var viewer = Add("Player", X - 1, Y); var events = new RenderProbe(); owner.AddPart(events);
            Assert.NotNull(owner.GetPart<DamageFlashPart>()); var hp = owner.GetStat("Hitpoints"); int beforeHP = hp.BaseValue;
            using (var paint = new Paint(world, zone)) using (var portraits = Portraits())
            {
                paint.Step(); int beforeRender = events.Renders; Assert.Greater(beforeRender, 0);
                CombatSystem.ApplyDamage(owner, 0, null, zone); Assert.AreEqual(beforeHP, hp.BaseValue); Assert.Zero(events.Damages);
                CombatSystem.ApplyDamage(owner, 1, null, zone);
                Assert.AreEqual(beforeHP - 1, hp.BaseValue); Assert.Greater(hp.BaseValue, 0); Assert.AreEqual(1, events.Damages);
                Assert.AreEqual(beforeRender, events.Renders, "TakeDamage alone is not a Render event.");
                paint.Step(); Assert.AreEqual(beforeRender + 1, events.Renders, "Actual damage marks the selected cell dirty once.");
                Native(paint.At(X, Y), Resource(species)); Assert.AreEqual(QudColorParser.Parse("&R"), paint.Tint(X, Y));
                var document = Document(owner, viewer);
                for (int i = 0; i < 3; i++) { View(); Assert.AreSame(Resource(species), portraits.GetPortrait(document, paint.Renderer).Texture); }
                paint.Dirty(X + 1, Y); paint.Step(); paint.Step();
                Assert.AreEqual(beforeRender + 1, events.Renders, "Neighbor reclaim/quiet frames/readers must not spend hurt frames.");
                Assert.AreEqual(QudColorParser.Parse("&R"), paint.Tint(X, Y));
                for (int frame = 2; frame <= 4; frame++)
                {
                    paint.Dirty(X, Y); paint.Step(); Assert.AreEqual(beforeRender + frame, events.Renders);
                    Native(paint.At(X, Y), Resource(species)); Assert.AreEqual(frame <= 3 ? QudColorParser.Parse("&R") : Color.white, paint.Tint(X, Y));
                }
                Assert.AreEqual(beforeHP - 1, hp.BaseValue); Assert.AreEqual(1, events.Damages);
            }
        }

        [TestCase("visual-id")] [TestCase("variant")] [TestCase("glyph-variants")] [TestCase("glyph")]
        [TestCase("foreign-render")] [TestCase("duplicate-render")] [TestCase("foreign-physics")] [TestCase("missing-physics")]
        [TestCase("foreign-body")] [TestCase("missing-body")] [TestCase("foreign-brain")] [TestCase("duplicate-brain")]
        [TestCase("hidden")] [TestCase("creature-tag")] [TestCase("item-tag")] [TestCase("player-tag")]
        [TestCase("carried")] [TestCase("equipped")] [TestCase("takeable")] [TestCase("footprint")]
        [TestCase("dead")] [TestCase("empty-stack")] [TestCase("foreign-stack")]
        public void MalformedOrRepurposedKnownOwnerCannotBorrowTheLivingBody(string fault)
        {
            var textures = Textures(); using (var art = Construct(textures))
            {
                var query = Resolver(art); var owner = Owner("GlasspaneFrog"); Assert.True(query(owner, out var initial)); Native(initial, textures[1]);
                var render = owner.GetPart<RenderPart>(); var physics = owner.GetPart<PhysicsPart>();
                switch (fault)
                {
                    case "visual-id": render.VisualID = "custom"; break;
                    case "variant": render.VisualVariant = "custom"; break;
                    case "glyph-variants": render.GlyphVariants = "fx"; break;
                    case "glyph": render.RenderString = "Q"; break;
                    case "foreign-render": render.ParentEntity = new Entity(); break;
                    case "duplicate-render": owner.AddPart(new RenderPart { RenderString = "f" }); break;
                    case "foreign-physics": physics.ParentEntity = new Entity(); break;
                    case "missing-physics": Assert.True(owner.RemovePart(physics)); break;
                    case "foreign-body": owner.GetPart<Body>().ParentEntity = new Entity(); break;
                    case "missing-body": Assert.True(owner.RemovePart(owner.GetPart<Body>())); break;
                    case "foreign-brain": owner.GetPart<BrainPart>().ParentEntity = new Entity(); break;
                    case "duplicate-brain": owner.AddPart(new BrainPart()); break;
                    case "hidden": render.Visible = false; break;
                    case "creature-tag": owner.Tags.Remove("Creature"); break;
                    case "item-tag": owner.SetTag("Item"); break;
                    case "player-tag": owner.SetTag("Player"); break;
                    case "carried": physics.InInventory = new Entity(); break;
                    case "equipped": physics.Equipped = new Entity(); break;
                    case "takeable": physics.Takeable = true; break;
                    case "footprint": owner.AddPart(new SpatialFootprintPart { CellsRaw = "0,0;1,0" }); break;
                    case "dead": owner.GetStat("Hitpoints").BaseValue = 0; break;
                    case "empty-stack": owner.AddPart(new StackerPart { StackCount = 0 }); break;
                    case "foreign-stack": var stack = new StackerPart(); owner.AddPart(stack); stack.ParentEntity = new Entity(); break;
                    default: Assert.Fail(fault); break;
                }
                Assert.False(query(owner, out var refused), fault); Assert.IsNull(refused);
            }
        }

        sealed class RenderProbe : Part
        {
            public int Renders, Damages;
            public override bool HandleEvent(GameEvent e) { if (e.ID == "Render") Renders++; if (e.ID == "TakeDamage") Damages++; return true; }
        }
        sealed class Paint : IDisposable
        {
            readonly HotbarSaveFixture saved;
            readonly GameObject root;
            readonly Tilemap overlay, main, ground;
            public ZoneRenderer Renderer { get; }
            readonly Action<int, int, string> oldCell = ZoneRenderHooks.CellDirtyCallback;
            readonly Action<string> oldFull = ZoneRenderHooks.FullDirtyCallback;
            readonly FieldInfo bound = typeof(ZoneTileStateSystem).GetField("_boundRenderZone", S);
            readonly object oldBound; readonly Zone previous; readonly Action<int, int> oldChanged; readonly Action oldSight;
            readonly Dictionary<FieldInfo, object> perf = new Dictionary<FieldInfo, object>();
            public Paint(FlatPixelGreyboxCatalog catalog, Zone z)
            {
                oldBound = bound.GetValue(null);
                if (oldBound is WeakReference<Zone> weak && weak.TryGetTarget(out previous)) { oldChanged = previous.TileState.OnCellChanged; oldSight = previous.TileState.OnSightChanged; }
                try
                {
                    foreach (var field in typeof(EnvironmentSpriteRenderer.Perf).GetFields(BindingFlags.Public | BindingFlags.Static)) perf[field] = field.GetValue(null);
                    saved = new HotbarSaveFixture(); root = new GameObject("Stump fauna submitted grid", typeof(Grid)); saved.Root.transform.SetParent(root.transform, false);
                    main = saved.Root.GetComponent<Tilemap>(); var bg = new GameObject("Background", typeof(Tilemap)); bg.transform.SetParent(root.transform, false); ground = bg.GetComponent<Tilemap>();
                    var marks = new GameObject("Hazards", typeof(Tilemap)); marks.transform.SetParent(root.transform, false);
                    Renderer = saved.Renderer; Write(Renderer, "_tilemap", main); Write(Renderer, "_bgTilemap", ground); Write(Renderer, "_tileStateTilemap", marks.GetComponent<Tilemap>());
                    var camera = new GameObject("Camera", typeof(Camera)); camera.transform.SetParent(root.transform, false); camera.GetComponent<Camera>().orthographic = true; Write(Renderer, "_mainCamera", camera.GetComponent<Camera>());
                    var env = saved.Root.AddComponent<EnvironmentSpriteRenderer>(); env.Init(root.transform, main, ground); env.RenderingEnabled = true; env.FlatPixelGreyboxEnabled = true;
                    Write(env, "_flatPixelGreybox", catalog); overlay = (Tilemap)Read(env, "_overlayTilemap"); Write(Renderer, "_envSpriteRenderer", env); Write(Renderer, "_flatPixelModeEnabled", true); Renderer.SetZone(z);
                    ZoneRenderHooks.CellDirtyCallback = Renderer.MarkCellDirty; ZoneRenderHooks.FullDirtyCallback = Renderer.MarkDirty;
                }
                catch { Dispose(); throw; }
            }
            static Vector3Int Position(int x, int y) => new Vector3Int(x, Zone.Height - 1 - y, 0);
            public Tile At(int x, int y) { var at = Position(x, y); return overlay.GetTile<Tile>(at) ?? main.GetTile<Tile>(at); }
            public Color Tint(int x, int y) { var at = Position(x, y); return overlay.HasTile(at) ? overlay.GetColor(at) : main.GetColor(at); }
            public Color Lit(Color color, int x, int y) { var light = (LightMap)Read(Renderer, "_lightMap"); return light == null ? color : light.ApplyToColor(color, x, y); }
            public Tile GroundAt(int x, int y) => ground.GetTile<Tile>(Position(x, y));
            public void Dirty(int x, int y) => Renderer.MarkCellDirty(x, y, "Stump fauna fixture");
            public void Step() { try { typeof(ZoneRenderer).GetMethod("LateUpdate", I).Invoke(Renderer, null); } catch (TargetInvocationException e) { throw e.InnerException ?? e; } }
            public void Dispose()
            {
                try { saved?.Dispose(); if (root != null) Object.DestroyImmediate(root); }
                finally
                {
                    ZoneRenderHooks.CellDirtyCallback = oldCell; ZoneRenderHooks.FullDirtyCallback = oldFull; bound.SetValue(null, oldBound);
                    if (previous != null) { previous.TileState.OnCellChanged = oldChanged; previous.TileState.OnSightChanged = oldSight; }
                    foreach (var entry in perf) entry.Key.SetValue(null, entry.Value);
                }
            }
        }
